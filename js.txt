(function () {
  let initialized = false;
  let displayedTotal = 0;
  let animationFrame = null;

  function readCount(value) {
    const text = String(value || "").trim().toLowerCase();
    const suffix = text.slice(-1);
    const match = text.match(/[\d.,]+/);

    if (!match) return 0;

    if (suffix === "k" || suffix === "m") {
      const decimal = parseFloat(match[0].replace(",", ".")) || 0;
      return Math.round(decimal * (suffix === "k" ? 1000 : 1000000));
    }

    return parseInt(match[0].replace(/\D/g, ""), 10) || 0;
  }

  function animateTotal(nextTotal) {
    const output = document.getElementById("total_count");
    if (!output || nextTotal === displayedTotal) return;

    const card = document.querySelector(".viewer-black");
    const startTotal = displayedTotal;
    const difference = nextTotal - startTotal;
    const startedAt = performance.now();
    const duration = 450;

    cancelAnimationFrame(animationFrame);

    function frame(now) {
      const progress = Math.min((now - startedAt) / duration, 1);
      const eased = 1 - Math.pow(1 - progress, 3);
      const current = Math.round(startTotal + difference * eased);

      output.textContent = current.toLocaleString("pt-BR");

      if (progress < 1) {
        animationFrame = requestAnimationFrame(frame);
      } else {
        displayedTotal = nextTotal;
      }
    }

    if (card) {
      card.classList.remove("count-changed");
      void card.offsetWidth;
      card.classList.add("count-changed");
    }

    animationFrame = requestAnimationFrame(frame);
  }

  function updateTotal() {
    const container = document.getElementById("platform_container");
    if (!container) return;

    const counters = container.querySelectorAll('[class$="-count"]');
    let total = 0;

    counters.forEach(function (counter) {
      total += readCount(counter.textContent);
    });

    animateTotal(total);
  }

  function initializeViewerCount() {
    if (initialized) return;

    const container = document.getElementById("platform_container");
    if (!container) return;

    initialized = true;

    const observer = new MutationObserver(updateTotal);
    observer.observe(container, {
      childList: true,
      subtree: true,
      characterData: true
    });

    updateTotal();
  }

  document.addEventListener("onLoad", initializeViewerCount);
  document.addEventListener("DOMContentLoaded", initializeViewerCount);

  if (document.readyState !== "loading") {
    initializeViewerCount();
  }
})();
